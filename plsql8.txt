CREATE OR REPLACE PROCEDURE pr2
(
 p_eno IN employ2.eno%TYPE,
 p_comm OUT NUMBER
)
AS
BEGIN
 SELECT sal * 0.10
 INTO p_comm
 FROM employ2
 WHERE eno = p_eno;
END;
/

SET SERVEROUTPUT ON;

DECLARE
 N employ2.eno%TYPE := &N;
 comm NUMBER;
BEGIN
 pr2(N, comm);
 DBMS_OUTPUT.PUT_LINE('Commission is ' || comm);
END;
/
